คำสั่ง SQL ที่ใช้ประมวลผลรายงาน Money Management (Backend Query)
-- 1. SQL สรุปรายได้รวมและจำนวนผู้เรียนทั้งหมด SELECT COUNT(*) AS total_transactions, COUNT(DISTINCT learner_id) AS total_paid_learners, SUM(amount_paid) AS total_revenue, COUNT(CASE WHEN status = 'IN_PROGRESS' THEN 1 END) AS active_studying_count, COUNT(CASE WHEN status = 'COMPLETED' THEN 1 END) AS completed_count FROM enrollment WHERE status != 'DROPPED'; -- 2. SQL สรุปรายได้แยกตามแต่ละคอร์สเรียน SELECT c.course_id, c.course_code, c.title AS course_title, c.level, c.price, COUNT(e.enrollment_id) AS total_enrolled, SUM(CASE WHEN e.status != 'DROPPED' THEN e.amount_paid ELSE 0 END) AS total_revenue FROM course c LEFT JOIN enrollment e ON c.course_id = e.course_id GROUP BY c.course_id, c.course_code, c.title, c.level, c.price ORDER BY total_revenue DESC; -- 3. SQL ดึงรายชื่อผู้เรียนที่จ่ายเงินแล้ว กำลังเรียนคอร์สอะไรอยู่ SELECT e.enrollment_id, e.enrollment_date, e.status, e.payment_method, e.amount_paid, CONCAT(l.first_name, ' ', l.last_name) AS learner_name, l.email, l.phone, c.course_code, c.title AS course_title FROM enrollment e JOIN learner l ON e.learner_id = l.learner_id JOIN course c ON e.course_id = c.course_id ORDER BY e.enrollment_date DESC;
